<!-- Key nh3 API (from research, verified hands-on) -->

nh3.clean(
    html,                          # str: input HTML
    tags=None,                     # Set[str] | None: allowed tags (None = nh3.ALLOWED_TAGS)
    clean_content_tags=None,       # Set[str] | None: tags whose content is also removed
    attributes=None,               # Dict[str, Set[str]] | None: allowed attrs per tag; '*' = wildcard
    attribute_filter=None,         # Callable | None — post-filter on allowed attrs
    strip_comments=True,           # bool: remove HTML comments
    link_rel='noopener noreferrer',# str | None: rel attr injected into <a> tags
    generic_attribute_prefixes=None, # Set[str]: allow attrs with prefixes (e.g. {'data-'})
    url_schemes=None,              # Set[str] | None: allowed URL schemes
)

nh3.ALLOWED_TAGS   # 75 tags (NOT 118 like bleach generally_xss_safe)
nh3.ALLOWED_ATTRIBUTES  # Very restrictive defaults

Key behavioral differences from bleach:
- nh3 REMOVES <script>/<style> tag content entirely (bleach only strips tags, keeps content)
- nh3 adds rel="noopener noreferrer" to <a> tags by default (set link_rel=None to disable)
- nh3 has no "allow all attributes via callable" — must enumerate explicitly
- nh3 with tags=frozenset(), attributes={} strips all tags but preserves content (same as bleach)
- Adding 'style' to nh3 tags set causes PanicException — style is always in clean_content_tags
- If 'rel' is in attributes for 'a' AND link_rel is set, nh3 panics

  **New implementation functions:**

  ```python
  def xss_protection_nh3(html: str) -> str:
      """Sanitizer that removes tags not considered XSS safe.

      Uses nh3 with a curated attribute allowlist. Unlike the deprecated
      bleach-based xss_protection(), this does NOT allow all attributes —
      only explicitly listed safe attributes are preserved.

      Script and style tag content is removed entirely (not just the tags).
      Links get rel="noopener noreferrer" for security.

      :param html: HTML to be sanitized
      :type html: str
      :result: sanitized HTML
      :rtype: str
      """
      return nh3.clean(
          html,
          tags=_XSS_SAFE_TAGS,
          attributes=_XSS_SAFE_ATTRS,
          link_rel="noopener noreferrer",
      )


  def minimal_html_nh3(html: str) -> str:
      """Sanitizer that only leaves a basic set of tags and attributes.

      Based on markdown and print tag/attribute sets. Style attributes
      are NOT allowed (tightened from bleach which left empty style="").

      :param html: HTML to be sanitized
      :type html: str
      :result: sanitized HTML
      :rtype: str
      """
      return nh3.clean(
          html,
          tags=_MINIMAL_TAGS,
          attributes=_MINIMAL_ATTRS,
          link_rel="noopener noreferrer",
      )


  def no_html_nh3(html: str) -> str:
      """Sanitizer that removes **all** tags.

      :param html: HTML to be sanitized
      :type html: str
      :result: plain text
      :rtype: str
      """
      return nh3.clean(html, tags=frozenset(), attributes={})
  ```
